function formatDateTime() {
const now = new Date();
const options = {
year: 'numeric',
month: 'long',
day: 'numeric',
hour: '2-digit',
minute: '2-digit',
second: '2-digit',
hour12: false
};
return now.toLocaleDateString('th-TH', options);
}

function validateEmail(email) {
return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
}

function validatePhone(phone) {
return /^0\d{9}$/.test(phone);
}

function validateIdCard(idCard) {
return !idCard || /^\d{13}$/.test(idCard);
}

// ตัวแปรเก็บสถานะการเริ่มต้นระบบ
let systemInitialized = false;
let initializationInProgress = false;

async function initializeDefaultUsers() {
// ป้องกันการเรียกใช้ซ้ำ
if (systemInitialized || initializationInProgress) {
console.log('ระบบเริ่มต้นแล้วหรือกำลังดำเนินการอยู่ - ข้าม');
return;
}

// ตั้งค่าสถานะว่ากำลังเริ่มต้นระบบ
initializationInProgress = true;
console.log('เริ่มต้นการตรวจสอบผู้ใช้เริ่มต้น...');

try {
// รอให้ข้อมูลโหลดเสร็จก่อน
await new Promise(resolve => setTimeout(resolve, 1500));

const existingUsers = allUsers.filter(u => u.type === 'user');
console.log('พบผู้ใช้ในระบบ:', existingUsers.length, 'คน');

// ถ้ามีผู้ใช้ในระบบแล้ว ไม่ต้องสร้างเพิ่ม
if (existingUsers.length > 0) {
console.log('มีผู้ใช้ในระบบแล้ว - ไม่สร้างผู้ใช้เริ่มต้น');
systemInitialized = true;
return;
}

// สร้างผู้ใช้เริ่มต้นเฉพาะเมื่อไม่มีผู้ใช้เลยในระบบ
console.log('ไม่มีผู้ใช้ในระบบ - เริ่มสร้างผู้ใช้เริ่มต้น...');

for (const defaultUser of DEFAULT_USERS) {
const newUser = {
type: 'user',
...defaultUser,
id_card: '',
backup_email: '',
line_id: '',
created_at: new Date().toISOString()
};

const result = await window.dataSdk.create(newUser);
if (result.isOk) {
console.log('สร้างผู้ใช้สำเร็จ:', defaultUser.email);
} else {
console.error('ไม่สามารถสร้างผู้ใช้:', defaultUser.email);
}
}

console.log('สร้างผู้ใช้เริ่มต้นเสร็จสิ้น');
systemInitialized = true;

} catch (error) {
console.error('เกิดข้อผิดพลาดในการเริ่มต้นระบบ:', error);
} finally {
initializationInProgress = false;
console.log('สิ้นสุดกระบวนการเริ่มต้นระบบ');
}
}

<div class="flex items-center justify-between bg-white p-3 rounded-lg border border-purple-200">
<span class="text-gray-800">${school}</span>
<button type="button" data-index="${index}" class="remove-school-btn text-red-600 hover:text-red-800 p-1">
<svg class="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
<path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M6 18L18 6M6 6l12 12"/>
</svg>
</button>
</div>
`).join('');

supervisedSchoolsInput.value = schools.join(', ');

// เพิ่ม event listeners สำหรับปุ่มลบ
document.querySelectorAll('.remove-school-btn').forEach(btn => {
btn.addEventListener('click', (e) => {
const index = parseInt(e.currentTarget.dataset.index);
schools.splice(index, 1);
updateSchoolsList();
});
});
}

function addSchool() {
const schoolName = newSchoolInput.value.trim();
if (schoolName && !schools.includes(schoolName)) {
schools.push(schoolName);
newSchoolInput.value = '';
updateSchoolsList();
}
}

if (addSchoolBtn) {
addSchoolBtn.addEventListener('click', addSchool);
}

if (newSchoolInput) {
newSchoolInput.addEventListener('keypress', (e) => {
if (e.key === 'Enter') {
e.preventDefault();
addSchool();
}
});
}
}

function renderViewContent() {
switch (currentView) {
case 'home': return renderHome();
case 'profile': return renderProfile();
case 'assessment': return renderAssessment();
case 'history': return renderAssessmentHistory();
case 'report': return renderReport();
case 'membership': return renderMembershipInfo();
case 'internship': return renderInternshipInfo();
case 'dashboard': return currentUser.role === 'admin' ? renderDashboard() : renderHome();
case 'users': return currentUser.role === 'admin' ? renderUserManagement() : renderHome();
case 'settings': return currentUser.role === 'admin' ? renderSystemSettings() : renderHome();
default: return renderHome();
}
}

function render() {
const app = document.getElementById('app');
if (!app) return;

if (!currentUser) {
app.innerHTML = currentView === 'signup' ? renderSignup() : renderLogin();
attachAuthHandlers();
} else {
app.innerHTML = renderMainApp();
attachMainAppHandlers();
updateDateTime();
}
}

function attachAuthHandlers() {
const loginForm = document.getElementById('loginForm');
const signupForm = document.getElementById('signupForm');
const showSignupBtn = document.getElementById('showSignupBtn');
const backToLoginBtn = document.getElementById('backToLoginBtn');
const roleSelect = document.getElementById('roleSelect');

if (loginForm) {
loginForm.addEventListener('submit', async (e) => {
e.preventDefault();
const email = document.getElementById('loginEmail').value.trim();
const password = document.getElementById('loginPassword').value;

const loginBtn = document.getElementById('loginBtn');
loginBtn.disabled = true;
loginBtn.innerHTML = '<div class="spinner mx-auto"></div>';

const user = allUsers.find(u => u.type === 'user' && u.email === email && u.password === password);

if (!user) {
showToast('อีเมลหรือรหัสผ่านไม่ถูกต้อง', 'error');
loginBtn.disabled = false;
loginBtn.textContent = 'เข้าสู่ระบบ';
return;
}

if (user.status === 'pending') {
showToast('บัญชีของคุณรอการอนุมัติจากผู้ดูแลระบบ', 'info');
loginBtn.disabled = false;
loginBtn.textContent = 'เข้าสู่ระบบ';
return;
}

currentUser = user;
currentView = 'home';
assessmentState = { round: null, studentId: null, step: 'main' };
showToast('เข้าสู่ระบบสำเร็จ', 'success');
render();
});
}

if (signupForm) {
signupForm.addEventListener('submit', async (e) => {
e.preventDefault();

const fullName = document.getElementById('fullName').value.trim();
const idCard = document.getElementById('idCard').value.trim();
const email = document.getElementById('email').value.trim();
const backupEmail = document.getElementById('backupEmail').value.trim();
const password = document.getElementById('password').value;
const phone = document.getElementById('phone').value.trim();
const lineId = document.getElementById('lineId').value.trim();
const organization = document.getElementById('organization').value.trim();
const role = document.getElementById('roleSelect').value;

if (!validateEmail(email)) {
showToast('รูปแบบอีเมลไม่ถูกต้อง', 'error');
return;
}

if (!validatePhone(phone)) {
showToast('เบอร์โทรศัพท์ต้องเป็นตัวเลข 10 หลัก เริ่มต้นด้วย 0', 'error');
return;
}

if (!validateIdCard(idCard)) {
showToast('เลขบัตรประชาชนต้องเป็นตัวเลข 13 หลัก', 'error');
return;
}

const existingUser = allUsers.find(u => u.type === 'user' && u.email === email);
if (existingUser) {
showToast('อีเมลนี้ถูกใช้งานแล้ว', 'error');
return;
}

const signupBtn = document.getElementById('signupBtn');
signupBtn.disabled = true;
signupBtn.innerHTML = '<div class="spinner mx-auto"></div>';

const newUser = {
type: 'user',
full_name: fullName,
id_card: idCard,
email: email,
backup_email: backupEmail,
password: password,
phone: phone,
line_id: lineId,
organization: organization,
role: role,
status: 'pending',
created_at: new Date().toISOString()
};

if (role === 'student') {
newUser.student_id = document.getElementById('studentId')?.value || '';
newUser.major = document.getElementById('major')?.value || '';
newUser.internship_school = document.getElementById('internshipSchool')?.value || '';
newUser.teaching_subject = document.getElementById('teachingSubject')?.value || '';
newUser.start_date = document.getElementById('startDate')?.value || '';
} else if (role === 'instructor') {
newUser.supervised_schools = document.getElementById('supervisedSchools')?.value || '';
} else if (['mentor', 'school_admin', 'board_rep', 'institution'].includes(role)) {
newUser.affiliated_school = document.getElementById('affiliatedSchool')?.value || '';
}

const result = await window.dataSdk.create(newUser);

if (result.isOk) {
showToast('ลงทะเบียนสำเร็จ รอการอนุมัติจากผู้ดูแลระบบ', 'success');
currentView = 'login';
render();
} else {
showToast('เกิดข้อผิดพลาดในการลงทะเบียน', 'error');
signupBtn.disabled = false;
signupBtn.textContent = 'ลงทะเบียน';
}
});
}

if (showSignupBtn) {
showSignupBtn.addEventListener('click', () => {
currentView = 'signup';
render();
});
}

if (backToLoginBtn) {
backToLoginBtn.addEventListener('click', () => {
currentView = 'login';
render();
});
}

if (roleSelect) {
roleSelect.addEventListener('change', (e) => {
renderRoleSpecificFields(e.target.value);
});
}
}

function calculateScores() {
const form = document.getElementById('assessmentForm');
if (!form) return;

// Calculate section 1
let section1Total = 0;
let section1Count = 0;
for (let i = 1; i <= 12; i++) {
const selected = form.querySelector(`input[name="section1_q${i}"]:checked`);
if (selected) {
section1Total += parseInt(selected.value);
section1Count++;
}
}
const section1Average = section1Count > 0 ? (section1Total / section1Count).toFixed(2) : '0.00';

document.getElementById('section1Total').textContent = section1Total;
document.getElementById('section1Average').textContent = section1Average;

// Calculate section 2
let section2Total = 0;
let section2Count = 0;
for (let i = 1; i <= 9; i++) {
const selected = form.querySelector(`input[name="section2_q${i}"]:checked`);
if (selected) {
section2Total += parseInt(selected.value);
section2Count++;
}
}
const section2Average = section2Count > 0 ? (section2Total / section2Count).toFixed(2) : '0.00';

document.getElementById('section2Total').textContent = section2Total;
document.getElementById('section2Average').textContent = section2Average;

// Calculate section 3
let section3Total = 0;
let section3Count = 0;
for (let i = 1; i <= 20; i++) {
const selected = form.querySelector(`input[name="section3_q${i}"]:checked`);
if (selected) {
section3Total += parseInt(selected.value);
section3Count++;
}
}
const section3Average = section3Count > 0 ? (section3Total / section3Count).toFixed(2) : '0.00';

document.getElementById('section3Total').textContent = section3Total;
document.getElementById('section3Average').textContent = section3Average;

// Calculate overall totals
const totalScore = section1Total + section2Total + section3Total;
const totalCount = section1Count + section2Count + section3Count;
const overallAverage = totalCount > 0 ? (totalScore / totalCount).toFixed(2) : '0.00';

document.getElementById('totalScore').textContent = totalScore;
document.getElementById('overallAverage').textContent = overallAverage;
}
